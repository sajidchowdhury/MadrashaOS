   244→function generateStudents(): Array<{
   245→  code: string; name: string; name_bn: string; name_ar?: string;
   246→  classId: string; section: string; guardianId: string; roll: number;
   247→  gender: Gender; dob: string; admittedAt: string;
   248→}> {
   249→  const students: Array<{
   250→    code: string; name: string; name_bn: string; name_ar?: string;
   251→    classId: string; section: string; guardianId: string; roll: number;
   252→    gender: Gender; dob: string; admittedAt: string;
   253→  }> = [];
   254→  let roll = 1;
   255→  // Class 1 Section A — 8 students
   256→  for (let i = 0; i < 8; i++) {
   257→    students.push(makeStudent(i, roll++, IDS.class1, "A", "g-1"));
   258→  }
   259→  // Class 3 Section A — 10 students
   260→  for (let i = 8; i < 18; i++) {
   261→    students.push(makeStudent(i, roll++, IDS.class3, "A", "g-2"));
   262→  }
   263→  // Class 5 Section A — 12 students (the famous Class 5-A)
   264→  for (let i = 18; i < 30; i++) {
   265→    students.push(makeStudent(i, roll++, IDS.class5, "A", "g-3"));
   266→  }
   267→  // Class 5 Section B — 10 students
   268→  for (let i = 30; i < 40; i++) {
   269→    students.push(makeStudent(i, roll++, IDS.class5, "B", "g-4"));
   270→  }
   271→  return students;
   272→}
   273→
   274→function makeStudent(
   275→  i: number, roll: number, classId: string, section: string, _guardianKey: string,
   276→) {
   277→  const gender: Gender = i % 2 === 0 ? "male" : "female";
   278→  const firstEn = FIRST_NAMES_EN[i % FIRST_NAMES_EN.length];
   279→  const firstBn = FIRST_NAMES_BN[i % FIRST_NAMES_BN.length];
   280→  const surname = SURNAMES[i % SURNAMES.length];
   281→  const surnameBn = SURNAMES_BN[i % SURNAMES_BN.length];
   282→  const year = 2010 + (i % 6);
   283→  const month = ((i % 12) + 1).toString().padStart(2, "0");
   284→  const day = ((i % 28) + 1).toString().padStart(2, "0");
   285→  return {
   286→    code: `MOS-2026-${(i + 1).toString().padStart(3, "0")}`,
   287→    name: `${firstEn} ${surname}`,
   288→    name_bn: `${firstBn} ${surnameBn}`,
   289→    name_ar: i % 4 === 0 ? `${firstEn} بن ${surname}` : undefined,
   290→    classId,
   291→    section,
   292→    guardianId: _guardianKey,
   293→    roll,
   294→    gender,
   295→    dob: `${year}-${month}-${day}`,
   296→    admittedAt: "2026-01-15",
   297→  };
   298→}
